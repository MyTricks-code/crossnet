package server

import (
	"log"
	"net"
	"fileTransfer/utils"
)

func Start(){
	listener, err := net.Listen("tcp", "0.0.0.0:8080")
	if err!=nil {
		log.Fatal("[Server] Failed to start", err);
	}
	defer listener.Close()

	log.Println("Waiting for connections...")
	conn, err:=listener.Accept()
	if err!=nil{
		log.Printf("Connection missed\n");
		return
	}
	
	utils.Decoder(conn)
	
	_, err = conn.Write([]byte("Hello from server, file saved"))
	if err!=nil{
		log.Println("Error sending data from server: ", err)
	}
	defer conn.Close()
}
